+def generate_knot_distribution(
+    K: int = 60,
+    samples_per_marginal: int = 10,
+    std: float = 0.1,
+    seed: int | None = None,
+) -> torch.Tensor:
+    """Generate synthetic Knot distribution from ALI-CFM Appendix B.
+
+    Returns a torch.Tensor of shape [K, samples_per_marginal, 2].
+    """
+    if K <= 0 or K % 3 != 0:
+        raise ValueError(f"K (number of marginals) must be a positive multiple of 3, got {K}")
+
+    t = np.linspace(0.0, 3.0, K)
+    # \tilde{t} = 3 * (t / t_max) - 1.5 where t_max = 3
+    t_tilde = t - 1.5
+
+    seg_size = K // 3
+    t1 = t_tilde[:seg_size]
+    t2 = t_tilde[seg_size : 2 * seg_size]
+    t3 = t_tilde[2 * seg_size :]
+
+    # \mu_X(t)
+    mu_x1 = 3.0 * (t1 + 0.5)
+    mu_x2 = np.cos(2.0 * np.pi * (t2 - 0.75))
+    mu_x3 = 3.0 * (t3 - 0.5)
+    mu_x = np.concatenate([mu_x1, mu_x2, mu_x3])
+
+    # \mu_Y(t)
+    mu_y1 = -0.5 * np.tanh(5.0 * (t1 + 1.0)) + 0.5
+    mu_y2 = np.sin(2.0 * np.pi * (t2 - 0.75))
+    mu_y3 = 0.5 * np.tanh(5.0 * (t3 - 1.0)) + 0.5
+    mu_y = np.concatenate([mu_y1, mu_y2, mu_y3])
+
+    rng = np.random.default_rng(seed)
+    x_samples = mu_x[:, None] + rng.normal(0.0, std, size=(K, samples_per_marginal))
+    y_samples = mu_y[:, None] + rng.normal(0.0, std, size=(K, samples_per_marginal))
+
+    samples = np.stack([x_samples, y_samples], axis=-1)
+    return torch.from_numpy(samples).float()
+

+class MultiMarginalDataset(Dataset):
+    """Dataset wrapping multi-marginal snapshots [K, N, D].
+
+    Flattened across marginals while retaining the temporal k_index (0 to K-1).
+    """
+
+    def __init__(self, data: torch.Tensor):
+        if data.ndim != 3:
+            raise ValueError(f"expected 3D tensor [K, N, D], got shape {tuple(data.shape)}")
+        self.data = data
+        self.K = data.shape[0]
+        self.N = data.shape[1]
+        self.D = data.shape[2]
+
+    def __len__(self) -> int:
+        return self.K * self.N
+
+    def __getitem__(self, index: int) -> tuple[torch.Tensor, torch.Tensor]:
+        k_index = index // self.N
+        n_index = index % self.N
+        return self.data[k_index, n_index], torch.tensor(k_index, dtype=torch.long)

+    """Wrapper adapting ALI-CFM velocity networks to the Flow2PC teacher interface.
+
+    Accepts calls in the standard Flow2PC format: `teacher(time, path_sample)`
+    where `time` is global in [0, 1] and maps it to the interval I_k expected by ALI-CFM.
+    """
+
+    def __init__(
+        self,

+    ):
+        super().__init__()
+        self.num_marginals = num_marginals
+        self.map_to_interval = map_to_interval
+        if isinstance(checkpoint_path, nn.Module):

+
+        path = Path(checkpoint_path)
+        try:

+        except Exception:

+
+        if isinstance(checkpoint, nn.Module):
+            self.velocity_net = checkpoint
+        elif isinstance(checkpoint, dict):
+            if (
+                checkpoint.get("time_format") == "global"
+                or checkpoint.get("model_config", {}).get("time_format") == "global"
+            ):
+                self.map_to_interval = False
+            # Check for module in common keys
+            for key in ("ot_cfm_model", "velocity_net", "model", "net"):
+                if key in checkpoint and isinstance(checkpoint[key], nn.Module):
+                    self.velocity_net = checkpoint[key]
+                    return
+
+            # Check for state dict in common keys
+            state_dict = None
+            for key in ("ot_cfm_model", "velocity_net", "model", "net", "state_dict"):
+                if key in checkpoint and isinstance(checkpoint[key], dict):
+                    state_dict = checkpoint[key]
+                    break
+            if state_dict is None:
+                state_dict = checkpoint
+
+            self.velocity_net = self._build_mlp_from_state_dict(state_dict)
+            self.velocity_net.load_state_dict(state_dict)
+        else:
+            raise TypeError(f"unexpected checkpoint type: {type(checkpoint)}")
+
+    @staticmethod
+    def _build_mlp_from_state_dict(state_dict: dict) -> nn.Module:
+        """Reconstruct MLP architecture from state dict layer weights."""
+        weight_keys = [k for k in state_dict if k.endswith(".weight")]
+        weight_keys.sort()
+        if not weight_keys:
+            raise ValueError("could not find weight keys in state dict")
+
+        layers = []
+        for i, key in enumerate(weight_keys):
+            weight = state_dict[key]
+            out_features, in_features = weight.shape
+            linear = nn.Linear(in_features, out_features)
+            layers.append(linear)
+            if i < len(weight_keys) - 1:
+                layers.append(nn.SELU())
+
+        # If original state dict had 'net.' prefix, wrap in a module that has 'net' attribute
+        has_net_prefix = any(k.startswith("net.") for k in state_dict)
+        sequential = nn.Sequential(*layers)
+        if has_net_prefix:
+            class PrefixedMLP(nn.Module):
+                def __init__(self, net):
+                    super().__init__()
+                    self.net = net
+                def forward(self, x):
+                    return self.net(x)
+            return PrefixedMLP(sequential)
+        return sequential
+
+    def forward(self, time: torch.Tensor, path_sample: torch.Tensor) -> torch.Tensor:
+        """Evaluate velocity field given global time in [0, 1] and path samples."""
+        if time.ndim == 0:
+            time = time.unsqueeze(0)
+        time_col = time[:, None] if time.ndim == 1 else time
+        time_col = time_col.to(device=path_sample.device, dtype=path_sample.dtype)
+
+        K = self.num_marginals
+        k_indices = torch.floor(time_col * (K - 1)).clamp(0, K - 2)
+        t_lower = k_indices / (K - 1)
+        t_upper = (k_indices + 1.0) / (K - 1)
+        tau = (time_col - t_lower) / (t_upper - t_lower)
+        k_norm = k_indices / (K - 1.0)
+
+        flat_sample = path_sample.flatten(1)
+
+        # Inspect expected input dimension if available
+        in_features = getattr(self.velocity_net, "in_features", None)
+        if in_features is None:
+            net_obj = getattr(self.velocity_net, "net", self.velocity_net)
+            if isinstance(net_obj, (nn.Sequential, list, tuple)) and len(net_obj) > 0:
+                in_features = getattr(net_obj[0], "in_features", None)
+
+        if in_features == flat_sample.shape[-1] + 2:
+            # 2 time inputs: local progress tau and normalized discrete interval k_norm
+            cat_input = torch.cat([flat_sample, tau, k_norm], dim=-1)
+        elif in_features == flat_sample.shape[-1] + 3:
+            cat_input = torch.cat([flat_sample, tau, k_norm, time_col], dim=-1)
+        elif self.map_to_interval:
+            cat_input = torch.cat([flat_sample, tau], dim=-1)
+        else:
+            cat_input = torch.cat([flat_sample, time_col], dim=-1)
+
+        try:
+            return self.velocity_net(cat_input)
+        except TypeError:
+            try:
+                return self.velocity_net(time_input, path_sample)
+            except TypeError:
+                return self.velocity_net(path_sample, time_input)

+@torch.no_grad()
+def integrate_rk4(
+    teacher: ALICFMTeacherWrapper,
+    x0: torch.Tensor,
+    num_steps: int = 100,
+    device: torch.device = torch.device("cpu"),
+) -> tuple[torch.Tensor, torch.Tensor]:
+    """Integrate dx/dt = v(t, x) from t=0 to t=1 using Runge-Kutta 4th order."""
+    dt = 1.0 / num_steps
+    time_steps = torch.linspace(0.0, 1.0, num_steps + 1, device=device)
+    
+    trajectories = [x0.clone()]
+    current_x = x0.clone()
+
+    for i in range(num_steps):
+        t_curr = time_steps[i].repeat(current_x.shape[0])
+        t_mid = (time_steps[i] + 0.5 * dt).repeat(current_x.shape[0])
+        t_next = time_steps[i + 1].repeat(current_x.shape[0])
+
+        k1 = teacher(t_curr, current_x)
+        k2 = teacher(t_mid, current_x + 0.5 * dt * k1)
+        k3 = teacher(t_mid, current_x + 0.5 * dt * k2)
+        k4 = teacher(t_next, current_x + dt * k3)
+
+        current_x = current_x + (dt / 6.0) * (k1 + 2.0 * k2 + 2.0 * k3 + k4)
+        trajectories.append(current_x.clone())
+
+    # [num_steps + 1, batch_size, 2] -> [batch_size, num_steps + 1, 2]
+    traj_tensor = torch.stack(trajectories, dim=1)
+    return time_steps, traj_tensor

+import argparse
+import sys
+from pathlib import Path
+
+# Ensure local repository root takes precedence over any installed site-packages
+_REPO_ROOT = str(Path(__file__).resolve().parent.parent)
+if _REPO_ROOT not in sys.path or sys.path[0] != _REPO_ROOT:
+    sys.path.insert(0, _REPO_ROOT)
+
+import torch
+from torch import nn
+from torchcfm.optimal_transport import OTPlanSampler
+from repc.data import generate_knot_distribution
+
+
+class KnotVelocityMLP(nn.Module):
+    """Exact ALI-CFM paper architecture: 3 hidden layers of 256 units with SELU activations.
+
+    Input (4 dimensions):
+      - x[..., :2]: 2D spatial coordinates (x_t, y_t)
+      - x[..., 2:3]: local intra-interval time progress tau in [0, 1]
+      - x[..., 3:4]: normalized discrete interval identifier k / (K - 1) in [0, 1]
+    Output (2 dimensions): continuous velocity field dx/dt
+    """
+
+    def __init__(self, in_dim: int = 4, out_dim: int = 2, hidden_dim: int = 256):
+        super().__init__()
+        self.net = nn.Sequential(
+            nn.Linear(in_dim, hidden_dim),
+            nn.SELU(),
+            nn.Linear(hidden_dim, hidden_dim),
+            nn.SELU(),
+            nn.Linear(hidden_dim, hidden_dim),
+            nn.SELU(),
+            nn.Linear(hidden_dim, out_dim),
+        )
+
+    def forward(self, x: torch.Tensor) -> torch.Tensor:
+        return self.net(x)
+
+
+def parse_args():
+    parser = argparse.ArgumentParser(description="Train a Knot ALI-CFM teacher checkpoint")
+    parser.add_argument("--output", type=Path, default=Path("checkpoints/knot_alicfm_teacher.pt"))
+    parser.add_argument("--steps", type=int, default=15000)
+    parser.add_argument("--batch-size", type=int, default=256)
+    parser.add_argument("--lr", type=float, default=1e-3)
+    parser.add_argument("--hidden-dim", type=int, default=256)
+    parser.add_argument("--K", type=int, default=60)
+    parser.add_argument("--samples-per-marginal", type=int, default=500)
+    parser.add_argument("--device", default="auto")
+    return parser.parse_args()
+
+
+def precompute_ot_couplings(data: torch.Tensor, device: torch.device) -> tuple[torch.Tensor, torch.Tensor]:
+    """Precompute exact Optimal Transport plan pairings for all K-1 transitions."""
+    K, N, D = data.shape
+    ot_sampler = OTPlanSampler(method="exact")
+    coupled_x0 = []
+    coupled_x1 = []
+
+    print(f"Precomputing exact OT plans across {K - 1} marginal transitions ({N} samples each)...")
+    for k in range(K - 1):
+        m0 = data[k]
+        m1 = data[k + 1]
+        p0, p1 = ot_sampler.sample_plan(m0, m1)
+        coupled_x0.append(p0)
+        coupled_x1.append(p1)
+
+    # Shape: [K-1, N, D]
+    x0_all = torch.stack(coupled_x0, dim=0).to(device)
+    x1_all = torch.stack(coupled_x1, dim=0).to(device)
+    return x0_all, x1_all
+
+
+def main():
+    args = parse_args()
+    device = torch.device(
+        ("cuda" if torch.cuda.is_available() else "cpu")
+        if args.device == "auto"
+        else args.device
+    )
+
+    args.output.parent.mkdir(parents=True, exist_ok=True)
+    model = KnotVelocityMLP(in_dim=4, out_dim=2, hidden_dim=args.hidden_dim).to(device)
+    optimizer = torch.optim.AdamW(model.parameters(), lr=args.lr, weight_decay=1e-5)
+    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(
+        optimizer, T_max=args.steps, eta_min=1e-6
+    )
+
+    print(f"Generating Knot distribution (K={args.K}, samples_per_marginal={args.samples_per_marginal})...")
+    raw_data = generate_knot_distribution(K=args.K, samples_per_marginal=args.samples_per_marginal)
+    # raw_data: [K, N, 2]
+    K = args.K
+    x0_coupled, x1_coupled = precompute_ot_couplings(raw_data, device=device)
+    num_transitions, num_samples, _ = x0_coupled.shape
+
+    print(f"Training Knot teacher for {args.steps} steps on {device}...")
+    loss_ema = None
+    log_interval = max(args.steps // 10, 100)
+
+    for step in range(1, args.steps + 1):
+        # Sample random transition intervals k in [0, K-2] and sample indices in [0, N-1]
+        k_indices = torch.randint(0, num_transitions, (args.batch_size,), device=device)
+        sample_indices = torch.randint(0, num_samples, (args.batch_size,), device=device)
+
+        x0 = x0_coupled[k_indices, sample_indices]  # [B, 2]
+        x1 = x1_coupled[k_indices, sample_indices]  # [B, 2]
+
+        # Sample local progress tau in [0, 1] within the interval
+        tau = torch.rand(args.batch_size, 1, device=device)
+        xt = (1.0 - tau) * x0 + tau * x1
+
+        # Discrete interval identifier k normalized in [0, 1] and continuous velocity u_t = dx/dt
+        k_norm = k_indices.unsqueeze(1).float() / (K - 1.0)
+        ut = (K - 1.0) * (x1 - x0)
+
+        # Predict velocity field conditioned on (x_t, tau, k_norm)
+        cat_input = torch.cat([xt, tau, k_norm], dim=-1)
+        vt = model(cat_input)
+        loss = nn.functional.mse_loss(vt, ut)
+
+        optimizer.zero_grad()
+        loss.backward()
+        optimizer.step()
+        scheduler.step()
+
+        loss_val = loss.item()
+        loss_ema = loss_val if loss_ema is None else 0.95 * loss_ema + 0.05 * loss_val
+
+        if step % log_interval == 0 or step == args.steps:
+            current_lr = scheduler.get_last_lr()[0]
+            print(
+                f"Step {step:5d}/{args.steps} | "
+                f"Loss: {loss_val:.6f} (EMA: {loss_ema:.6f}) | "
+                f"LR: {current_lr:.2e}",
+                flush=True,
+            )
+
+    # Save checkpoint in ALI-CFM format matching paper specifications
+    checkpoint = {
+        "model_type": "flow",
+        "time_format": "interval_and_k",
+        "model_config": {
+            "name": "alicfm",
+            "checkpoint_path": str(args.output.resolve()),
+            "num_marginals": args.K,
+            "hidden_dim": args.hidden_dim,
+            "in_dim": 4,
+            "activation": "selu",
+        },
+        "ot_cfm_model": model.state_dict(),
+        "flow_matching_config": {"name": "exact_ot_cfm", "sigma": 0.0},
+    }
+    torch.save(checkpoint, args.output)
+    print(f"Saved teacher checkpoint to {args.output}", flush=True)
+
+
+if __name__ == "__main__":
+    main()
